import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.databind.node.ObjectNode;
import org.finos.legend.engine.language.pure.grammar.from.PureGrammarParser;
import org.finos.legend.engine.shared.core.operational.errorManagement.EngineException;

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.List;
import java.util.concurrent.atomic.AtomicInteger;
import java.util.concurrent.atomic.AtomicReferenceArray;

/**
 * Probe source, never built in the repo (ruling 17, docs/rust-port/ENGINE_GAP_2026_11_09.md): every case through
 * legend-engine's own parser, one JVM, a pool of workers, each ask on a thread of its own with a big stack and a time
 * limit. Accept means the parser returned (the answer is not serialized: the scope is accept/refuse and place).
 * Usage: GapEngineProbe cases.jsonl out.jsonl threads timeoutSeconds
 */
public class GapEngineProbe {
    interface P {
        Object run() throws Exception;
    }

    static final ObjectMapper PLAIN = new ObjectMapper();
    static final long STACK = 512L << 20;

    record Case(String id, String kind, String text) {}

    static ObjectNode ask(P p) {
        ObjectNode r = PLAIN.createObjectNode();
        try {
            p.run();
            r.put("verdict", "ACCEPT");
        } catch (Throwable t) {
            r.put("verdict", "REJECT");
            r.put("class", t.getClass().getName());
            String m = String.valueOf(t.getMessage());
            r.put("message", m.length() > 600 ? m.substring(0, 600) : m);
            if (t instanceof EngineException ee) {
                var si = ee.getSourceInformation();
                if (si != null) {
                    r.put("line", si.startLine);
                    r.put("column", si.startColumn);
                    r.put("endLine", si.endLine);
                    r.put("endColumn", si.endColumn);
                }
                r.put("type", String.valueOf(ee.getErrorType()));
            }
        }
        return r;
    }

    /** One ask on a thread of its own; a thread that outlives the limit is abandoned and answered TIMEOUT. */
    static ObjectNode bounded(P p, long timeoutMs, boolean[] abandoned) throws InterruptedException {
        ObjectNode[] box = new ObjectNode[1];
        Thread t = new Thread(null, () -> box[0] = ask(p), "ask", STACK);
        t.setDaemon(true);
        t.start();
        t.join(timeoutMs);
        if (t.isAlive()) {
            abandoned[0] = true;
            ObjectNode r = PLAIN.createObjectNode();
            r.put("verdict", "TIMEOUT");
            return r;
        }
        return box[0];
    }

    public static void main(String[] a) throws Exception {
        List<Case> cases = new ArrayList<>();
        try (BufferedReader in = Files.newBufferedReader(Path.of(a[0]), StandardCharsets.UTF_8)) {
            String line;
            while ((line = in.readLine()) != null) {
                JsonNode c = PLAIN.readTree(line);
                cases.add(new Case(c.get("id").asText(), c.get("kind").asText(), c.get("text").asText()));
            }
        }
        int threads = Integer.parseInt(a[2]);
        long timeoutMs = Long.parseLong(a[3]) * 1000L;
        AtomicInteger next = new AtomicInteger();
        AtomicReferenceArray<String> results = new AtomicReferenceArray<>(cases.size());
        AtomicInteger timeouts = new AtomicInteger();
        for (int w = 0; w < threads; w++) {
            Thread worker = new Thread(() -> {
                PureGrammarParser parser = PureGrammarParser.newInstance();
                while (true) {
                    int i = next.getAndIncrement();
                    if (i >= cases.size()) {
                        return;
                    }
                    Case c = cases.get(i);
                    ObjectNode r = PLAIN.createObjectNode();
                    r.put("id", c.id());
                    r.put("kind", c.kind());
                    try {
                        boolean[] abandoned = {false};
                        PureGrammarParser pp = parser;
                        switch (c.kind()) {
                            case "model" -> r.set("model", bounded(() -> pp.parseModel(c.text()), timeoutMs, abandoned));
                            case "lambda" -> r.set("lambda", bounded(() -> pp.parseLambda(c.text()), timeoutMs, abandoned));
                            case "spec" -> {
                                r.set("spec", bounded(() -> pp.parseValueSpecification(c.text(), "", 0, 0, true), timeoutMs, abandoned));
                                r.set("lambda", bounded(() -> pp.parseLambda(c.text()), timeoutMs, abandoned));
                            }
                            default -> throw new IllegalArgumentException(c.kind());
                        }
                        if (abandoned[0]) {
                            timeouts.incrementAndGet();
                            parser = PureGrammarParser.newInstance();
                        }
                        results.set(i, PLAIN.writeValueAsString(r));
                    } catch (Exception e) {
                        throw new RuntimeException(e);
                    }
                }
            }, "worker-" + w);
            worker.setDaemon(true);
            worker.start();
        }
        long t0 = System.currentTimeMillis();
        try (BufferedWriter out = Files.newBufferedWriter(Path.of(a[1]), StandardCharsets.UTF_8)) {
            for (int i = 0; i < cases.size(); i++) {
                String s;
                while ((s = results.get(i)) == null) {
                    Thread.sleep(2);
                }
                out.write(s);
                out.newLine();
                if (i % 5000 == 0) {
                    out.flush();
                    System.err.println(i + "/" + cases.size() + " at " + (System.currentTimeMillis() - t0) / 1000 + "s, timeouts " + timeouts.get());
                }
            }
        }
        System.err.println("done " + cases.size() + " cases in " + (System.currentTimeMillis() - t0) / 1000 + "s, timeouts " + timeouts.get());
        System.exit(0);
    }
}
